void loop() {
  int x = analogRead(X_PIN);
  int y = analogRead(Y_PIN);
  bool pressed = (digitalRead(SW_PIN) == LOW);

  Serial.print("{\"jx\":");
  Serial.print(x);
  Serial.print(",\"jy\":");
  Serial.print(y);
  Serial.print(",\"btn\":");
  Serial.print(pressed ? "true" : "false");
  Serial.println("}");

}

import { useRef, useEffect, useState } from 'react';

export default function JoystickTestArena() {
  const canvasRef = useRef(null);
  const joystickRef = useRef({ jx: 512, jy: 512, btn: false });
  const prevBtnRef = useRef(false);

  // Game state
  const crosshairRef = useRef({ x: 400, y: 300 });
  const targetRef = useRef({ x: 250, y: 200, spawnTime: performance.now() });

  const [connected, setConnected] = useState(false);
  const [lastReactionMs, setLastReactionMs] = useState(null);
  const [score, setScore] = useState(0);

  // 1. Connect to Arduino via Web Serial
  const connectJoystick = async () => {
    try {
      const port = await navigator.serial.requestPort();
      await port.open({ baudRate: 115200 });
      setConnected(true);
      targetRef.current.spawnTime = performance.now();

      const textDecoder = new TextDecoderStream();
      port.readable.pipeTo(textDecoder.writable);
      const reader = textDecoder.readable.getReader();

      let buffer = '';
      while (true) {
        const { value, done } = await reader.read();
        if (done) break;

        buffer += value;
        const lines = buffer.split('\n');
        buffer = lines.pop();

        for (const line of lines) {
          if (line.trim()) {
            try {
              const data = JSON.parse(line);
              joystickRef.current = data; // Update ref directly for 60fps reading
            } catch (e) {
              // Ignore partial startup line
            }
          }
        }
      }
    } catch (err) {
      console.error('Failed to connect:', err);
    }
  };

  // Helper: Normalize 0-1023 to -1..1 with deadzone
  const normalizeAxis = (val) => {
    const centered = val - 512;
    if (Math.abs(centered) < 35) return 0;
    return Math.max(-1, Math.min(1, centered / 512));
  };

  // 2. 60fps Game Loop
  useEffect(() => {
    let animId;
    const canvas = canvasRef.current;
    const ctx = canvas.getContext('2d');

    const loop = () => {
      const { jx, jy, btn } = joystickRef.current;
      const vx = normalizeAxis(jx);
      const vy = normalizeAxis(jy);

      // Move crosshair (adjust '* 8' to change sensitivity)
      crosshairRef.current.x = Math.max(20, Math.min(canvas.width - 20, crosshairRef.current.x + vx * 8));
      crosshairRef.current.y = Math.max(20, Math.min(canvas.height - 20, crosshairRef.current.y + vy * 8));

      // Detect button click (rising edge: false -> true)
      if (btn && !prevBtnRef.current) {
        const dist = Math.hypot(
          crosshairRef.current.x - targetRef.current.x,
          crosshairRef.current.y - targetRef.current.y
        );

        // Hit detection (target radius 28 + crosshair radius 15)
        if (dist < 43) {
          const reactionTime = Math.round(performance.now() - targetRef.current.spawnTime);
          setLastReactionMs(reactionTime);
          setScore((s) => s + 1);

          // Spawn new target
          targetRef.current = {
            x: 60 + Math.random() * (canvas.width - 120),
            y: 60 + Math.random() * (canvas.height - 120),
            spawnTime: performance.now(),
          };
        }
      }
      prevBtnRef.current = btn;

      // Render background
      ctx.fillStyle = '#090d16';
      ctx.fillRect(0, 0, canvas.width, canvas.height);

      // Render Alien Target
      ctx.fillStyle = '#22c55e';
      ctx.beginPath();
      ctx.arc(targetRef.current.x, targetRef.current.y, 28, 0, Math.PI * 2);
      ctx.fill();

      // Render Player Crosshair (turns yellow when button is held)
      ctx.strokeStyle = btn ? '#facc15' : '#38bdf8';
      ctx.lineWidth = 3;
      ctx.beginPath();
      ctx.arc(crosshairRef.current.x, crosshairRef.current.y, 15, 0, Math.PI * 2);
      ctx.stroke();

      animId = requestAnimationFrame(loop);
    };

    animId = requestAnimationFrame(loop);
    return () => cancelAnimationFrame(animId);
  }, []);

  return (
    <div style={{ display: 'flex', flexDirection: 'column', alignItems: 'center', gap: '12px', padding: '20px', background: '#020617', color: 'white', minHeight: '100vh' }}>
      <h2>Neuro-Arcade Joystick Test</h2>
      
      {!connected ? (
        <button onClick={connectJoystick} style={{ padding: '10px 20px', fontSize: '16px', cursor: 'pointer' }}>
          Connect Arduino Joystick
        </button>
      ) : (
        <div style={{ display: 'flex', gap: '24px', fontSize: '18px' }}>
          <span>Targets Hit: <strong>{score}</strong></span>
          <span>Last Hand Reaction: <strong>{lastReactionMs ? `${lastReactionMs} ms` : '--'}</strong></span>
        </div>
      )}

      <canvas ref={canvasRef} width={800} height={500} style={{ border: '2px solid #334155', borderRadius: '8px' }} />
    </div>
  );
}
